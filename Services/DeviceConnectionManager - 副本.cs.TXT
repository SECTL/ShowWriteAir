namespace ShowWriteAir.Services

    /// <summary>
    /// 已连接设备信息项
    /// </summary>
    public class ConnectedDeviceInfo
    {
        public string Manufacturer { get; set; } = "Unknown";
        public string Model { get; set; } = "Unknown";
    }

    /// <summary>
    /// 设备连接管理器
    /// 负责管理设备连接、端口开放和二维码生成
    /// </summary>
    public class DeviceConnectionManager
    {
        /// <summary>
        /// 心跳超时时间（秒）。超过此时间未收到客户端任何消息（含 SWEC_PING）则判定连接断开。
        /// 客户端每 15 秒发送一次 SWEC_PING 作为心跳，此值应远大于客户端的 PING 间隔（取 30 秒 ≈ 2 个心跳周期）。
        /// </summary>
        private const int HeartbeatTimeoutSeconds = 30;

        private TcpListener _tcpListener;
        private int _port = 8888;
        private bool _isListening = false;
        private string _connectionCode;
        private BitmapImage _qrCodeImage;
        private readonly LanguageManager _languageManager;
        // 选定的网卡 Id（null 表示自动选择）
        private string _selectedAdapterId = null;

        // 已连接设备计数（线程安全）
        private int _connectedDeviceCount = 0;
        private readonly object _countLock = new object();

        public event Action<string> ConnectionStatusChanged;
        public event Action<string> ClientConnected;
        public event Action<byte[]> PhotoReceived;
        /// <summary>
        /// 已连接设备数变化时触发，参数为当前已连接设备数
        /// </summary>
        public event Action<int> ConnectedDeviceCountChanged;

        // 已连接设备信息表（按连接 ID 索引，线程安全）。设备型号在握手成功后由客户端通过 SWEC_DEVICE 上报。
        private readonly Dictionary<Guid, ConnectedDeviceInfo> _connectedDevices = new Dictionary<Guid, ConnectedDeviceInfo>();
        private readonly object _devicesLock = new object();

        /// <summary>
        /// 已连接设备信息列表变化时触发，参数为当前已连接设备信息的快照
        /// </summary>
        public event Action<List<ConnectedDeviceInfo>> ConnectedDevicesChanged;

        /// <summary>
        /// 获取当前已连接设备信息的快照
        /// </summary>
        public List<ConnectedDeviceInfo> GetConnectedDevices()
        {
            lock (_devicesLock)
            {
                return new List<ConnectedDeviceInfo>(_connectedDevices.Values);
            }
        }

        /// <summary>
        /// 触发已连接设备信息列表变化事件（在锁外调用，避免回调中访问管理器造成死锁）
        /// </summary>
        private void RaiseConnectedDevicesChanged()
        {
            var snapshot = GetConnectedDevices();
            ConnectedDevicesChanged?.Invoke(snapshot);
        }

        /// <summary>
        /// 连接状态
        /// </summary>
        public bool IsListening => _isListening;

        /// <summary>
        /// 当前已连接设备数
        /// </summary>
        public int ConnectedDeviceCount
        {
            get
            {
                lock (_countLock)
                {
                    return _connectedDeviceCount;
                }
            }
        }

        /// <summary>
        /// 增加已连接设备计数并通知
        /// </summary>
        private void IncrementConnectedCount()
        {
            int newCount;
            lock (_countLock)
            {
                _connectedDeviceCount++;
                newCount = _connectedDeviceCount;
            }
            Logger.Info("DeviceConnectionManager", $"设备已连接，当前连接数: {newCount}");
            ConnectedDeviceCountChanged?.Invoke(newCount);
        }

        /// <summary>
        /// 减少已连接设备计数并通知
        /// </summary>
        private void DecrementConnectedCount()
        {
            int newCount;
            lock (_countLock)
            {
                if (_connectedDeviceCount > 0)
                {
                    _connectedDeviceCount--;
                }
                newCount = _connectedDeviceCount;
            }
            Logger.Info("DeviceConnectionManager", $"设备已断开，当前连接数: {newCount}");
            ConnectedDeviceCountChanged?.Invoke(newCount);
        }

        /// <summary>
        /// 重置已连接设备计数（用于停止监听后）
        /// </summary>
        private void ResetConnectedCount()
        {
            int oldCount;
            lock (_countLock)
            {
                oldCount = _connectedDeviceCount;
                _connectedDeviceCount = 0;
            }
            bool devicesCleared = false;
            lock (_devicesLock)
            {
                if (_connectedDevices.Count > 0)
                {
                    _connectedDevices.Clear();
                    devicesCleared = true;
                }
            }
            if (oldCount > 0)
            {
                ConnectedDeviceCountChanged?.Invoke(0);
            }
            if (devicesCleared)
            {
                RaiseConnectedDevicesChanged();
            }
        }

        /// <summary>
        /// 端口号
        /// </summary>
        public int Port
        {
            get => _port;
            set => _port = value;
        }

        /// <summary>
        /// 选定网卡的 Id（null 表示自动选择）
        /// </summary>
        public string SelectedAdapterId
        {
            get => _selectedAdapterId;
            set => _selectedAdapterId = value;
        }

        /// <summary>
        /// 连接码
        /// </summary>
        public string ConnectionCode => _connectionCode;

        /// <summary>
        /// 二维码图像
        /// </summary>
        public BitmapImage QrCodeImage => _qrCodeImage;

        /// <summary>
        /// 构造函数
        /// </summary>
        public DeviceConnectionManager()
        {
            _languageManager = LanguageManager.Instance;
        }

        /// <summary>
        /// 开始监听连接
        /// </summary>
        public int FindAvailablePort(int startPort = 8888, int maxPort = 8999)
        {
            for (int port = startPort; port <= maxPort; port++)
            {
                try
                {
                    using (var socket = new Socket(AddressFamily.InterNetwork, SocketType.Stream, ProtocolType.Tcp))
                    {
                        socket.SetSocketOption(SocketOptionLevel.Socket, SocketOptionName.ReuseAddress, true);
                        socket.Bind(new IPEndPoint(IPAddress.Any, port));
                        return port;
                    }
                }
                catch (SocketException)
                {
                    Logger.Info("DeviceConnectionManager", $"端口 {port} 不可用");
                }
            }
            return -1;
        }

        public async Task StartListeningAsync()
        {
            try
            {
                if (_isListening)
                    return;

                _tcpListener = new TcpListener(IPAddress.Any, _port);
                _tcpListener.Server.SetSocketOption(SocketOptionLevel.Socket, SocketOptionName.ReuseAddress, true);
                _isListening = true;
                _tcpListener.Start();

                var message = _languageManager.GetTranslation("WaitingForConnection").Replace("...", "");
                ConnectionStatusChanged?.Invoke($"{message} {Environment.MachineName}...");

                Logger.Info("DeviceConnectionManager", $"开始监听端口 {_port}");

                while (_isListening)
                {
                    TcpClient client = await _tcpListener.AcceptTcpClientAsync();
                    await HandleClientAsync(client);
                }
            }
            catch (Exception ex)
            {
                _isListening = false;
                Logger.Error("DeviceConnectionManager", $"监听失败: {ex.Message}", ex);
                ConnectionStatusChanged?.Invoke($"连接失败: {ex.Message}");
            }
        }

        /// <summary>
        /// 停止监听
        /// </summary>
        public void StopListening()
        {
            try
            {
                _isListening = false;
                _tcpListener?.Stop();
                _tcpListener = null;

                // 停止监听后重置已连接设备计数
                ResetConnectedCount();

                ConnectionStatusChanged?.Invoke(_languageManager.GetTranslation("StoppedListening"));
                Logger.Info("DeviceConnectionManager", "已停止监听");
            }
            catch (Exception ex)
            {
                Logger.Error("DeviceConnectionManager", $"停止监听失败: {ex.Message}", ex);
            }
        }

        private async Task<string> ReadLineFromStreamAsync(NetworkStream stream, System.Threading.CancellationToken cancellationToken = default)
        {
            var buffer = new System.Collections.Generic.List<byte>();
            var singleByte = new byte[1];

            while (true)
            {
                int bytesRead = await stream.ReadAsync(singleByte, 0, 1, cancellationToken);
                if (bytesRead == 0) return null;

                if (singleByte[0] == '\n')
                {
                    break;
                }
                if (singleByte[0] != '\r')
                {
                    buffer.Add(singleByte[0]);
                }
            }

            return System.Text.Encoding.UTF8.GetString(buffer.ToArray());
        }

        private async Task SendLineAsync(NetworkStream stream, string line)
        {
            var data = System.Text.Encoding.UTF8.GetBytes(line + "\r\n");
            await stream.WriteAsync(data, 0, data.Length);
            await stream.FlushAsync();
        }

        /// <summary>
        /// 处理客户端连接
        /// </summary>
        private async Task HandleClientAsync(TcpClient client)
        {
            // 标记是否完成握手，只有在握手成功后才计入设备计数
            bool handshakeSuccess = false;
            // 当前连接的唯一标识，用于在设备信息表中增删条目
            var connectionId = Guid.NewGuid();
            try
            {
                Logger.Info("DeviceConnectionManager", "客户端已连接");
                ClientConnected?.Invoke(_languageManager.GetTranslation("DeviceConnected"));

                using (var stream = client.GetStream())
                {
                    var hello = await ReadLineFromStreamAsync(stream);
                    Logger.Info("DeviceConnectionManager", $"收到: {hello}");

                    if (hello == "SWEC_HELLO")
                    {
                        await SendLineAsync(stream, "SWEC_ACK");
                        ConnectionStatusChanged?.Invoke("握手中...");
                        Logger.Info("DeviceConnectionManager", "发送: SWEC_ACK");

                        var ready = await ReadLineFromStreamAsync(stream);
                        if (ready == "SWEC_READY")
                        {
                            ConnectionStatusChanged?.Invoke(_languageManager.GetTranslation("HandshakeSuccess"));
                            Logger.Info("DeviceConnectionManager", "收到: SWEC_READY，握手完成");
                            // 握手成功，计入已连接设备数
                            handshakeSuccess = true;
                            IncrementConnectedCount();
                            // 握手成功后登记设备信息条目（型号暂为 Unknown，待客户端上报 SWEC_DEVICE 后更新）
                            lock (_devicesLock)
                            {
                                _connectedDevices[connectionId] = new ConnectedDeviceInfo();
                            }
                            RaiseConnectedDevicesChanged();
                        }
                        else
                        {
                            Logger.Error("DeviceConnectionManager", $"握手失败，期望SWEC_READY，收到: {ready ?? "[空/断开]"}");
                            ConnectionStatusChanged?.Invoke($"握手失败: 收到无效响应");
                            return;
                        }
                    }
                    else
                    {
                        Logger.Error("DeviceConnectionManager", $"握手失败，收到: {hello ?? "[空/断开]"}");
                        ConnectionStatusChanged?.Invoke($"握手失败: 收到无效响应");
                        return;
                    }

                    string message;
                    while (true)
                    {
                        // 心跳超时检测：每次读取下一条消息时设置超时
                        // 客户端通过定期发送 SWEC_PING 维持心跳，超过 HeartbeatTimeoutSeconds 未收到任何消息则判定断开
                        using var heartbeatCts = new System.Threading.CancellationTokenSource(TimeSpan.FromSeconds(HeartbeatTimeoutSeconds));
                        try
                        {
                            message = await ReadLineFromStreamAsync(stream, heartbeatCts.Token);
                        }
                        catch (System.Threading.Tasks.TaskCanceledException)
                        {
                            Logger.Warning("DeviceConnectionManager", $"心跳超时：超过 {HeartbeatTimeoutSeconds} 秒未收到客户端消息（含 SWEC_PING），判定连接断开");
                            ConnectionStatusChanged?.Invoke("设备心跳超时，连接已断开");
                            break;
                        }

                        if (message == null)
                        {
                            Logger.Info("DeviceConnectionManager", "客户端正常断开连接");
                            break;
                        }

                        if (message == "PHOTO_BIN")
                        {
                            try
                            {
                                Logger.Info("DeviceConnectionManager", "收到二进制照片协议");

                                byte[] lenBuf = new byte[4];
                                int totalRead = 0;
                                while (totalRead < 4)
                                {
                                    int read = await stream.ReadAsync(lenBuf, totalRead, 4 - totalRead);
                                    if (read == 0)
                                    {
                                        Logger.Error("DeviceConnectionManager", "连接已断开，未能读取完整长度头");
                                        await SendLineAsync(stream, "PHOTO_ERROR");
                                        break;
                                    }
                                    totalRead += read;
                                }

                                if (totalRead < 4) continue;

                                int length = BinaryPrimitives.ReadInt32BigEndian(lenBuf);
                                Logger.Info("DeviceConnectionManager", $"照片数据长度: {length} 字节");

                                if (length <= 0 || length > 50 * 1024 * 1024)
                                {
                                    Logger.Error("DeviceConnectionManager", $"无效的照片长度: {length}");
                                    await SendLineAsync(stream, "PHOTO_ERROR");
                                    continue;
                                }

                                byte[] imgData = new byte[length];
                                int totalImgRead = 0;
                                while (totalImgRead < length)
                                {
                                    int read = await stream.ReadAsync(imgData, totalImgRead, length - totalImgRead);
                                    if (read == 0)
                                    {
                                        Logger.Error("DeviceConnectionManager", $"流已结束，已读取 {totalImgRead}/{length} 字节");
                                        break;
                                    }
                                    totalImgRead += read;
                                }

                                if (totalImgRead == length)
                                {
                                    // 异步派发照片事件，避免照片处理（含磁盘 IO 与 UI 操作）阻塞网络读取循环
                                    // 否则下一条 SWEC_PING 可能无法及时读取，触发 30 秒心跳超时误判断连
                                    var photoDataCopy = imgData;
                                    _ = Task.Run(() => PhotoReceived?.Invoke(photoDataCopy));
                                    ConnectionStatusChanged?.Invoke(_languageManager.GetTranslation("PhotoReceived"));
                                    await SendLineAsync(stream, "PHOTO_OK");
                                    Logger.Info("DeviceConnectionManager", "二进制照片接收完成");
                                }
                                else
                                {
                                    await SendLineAsync(stream, "PHOTO_ERROR");
                                    Logger.Error("DeviceConnectionManager", $"照片接收不完整，期望 {length} 字节，实际读取 {totalImgRead} 字节");
                                }
                            }
                            catch (Exception ex)
                            {
                                Logger.Error("DeviceConnectionManager", $"二进制照片接收异常: {ex.Message}", ex);
                                await SendLineAsync(stream, "PHOTO_ERROR");
                            }
                        }
                        else if (message.StartsWith("PHOTO:"))
                        {
                            Logger.Info("DeviceConnectionManager", $"收到Base64照片数据，长度: {message.Length} 字符");
                            var base64Data = message.Substring(6);
                            var photoData = Convert.FromBase64String(base64Data);
                            // 异步派发照片事件，避免照片处理阻塞网络读取循环（详见 PHOTO_BIN 分支注释）
                            _ = Task.Run(() => PhotoReceived?.Invoke(photoData));
                            ConnectionStatusChanged?.Invoke(_languageManager.GetTranslation("PhotoReceived"));

                            await SendLineAsync(stream, "PHOTO_OK");
                        }
                        else if (message == "SWEC_PING")
                        {
                            // 心跳：客户端每 15 秒发送一次 SWEC_PING，服务端立即回复 SWEC_PONG。
                            // SWEC_PING 频率较高，使用 Debug 级别避免日志刷屏
                            Logger.Debug("DeviceConnectionManager", "收到 SWEC_PING，回复 SWEC_PONG");
                            await SendLineAsync(stream, "SWEC_PONG");
                        }
                        else if (message.StartsWith("SWEC_DEVICE "))
                        {
                            // 客户端在握手成功后上报其设备型号：SWEC_DEVICE {"manufacturer":"<厂商>","model":"<型号>"}
                            var clientPayload = message.Substring("SWEC_DEVICE ".Length);
                            Logger.Info("DeviceConnectionManager", $"收到客户端设备型号: {clientPayload}");

                            // 解析并更新当前连接对应的设备信息条目
                            try
                            {
                                var info = JsonConvert.DeserializeObject<ConnectedDeviceInfo>(clientPayload);
                                if (info != null)
                                {
                                    lock (_devicesLock)
                                    {
                                        if (_connectedDevices.ContainsKey(connectionId))
                                        {
                                            _connectedDevices[connectionId].Manufacturer = string.IsNullOrWhiteSpace(info.Manufacturer) ? "Unknown" : info.Manufacturer;
                                            _connectedDevices[connectionId].Model = string.IsNullOrWhiteSpace(info.Model) ? "Unknown" : info.Model;
                                        }
                                    }
                                    RaiseConnectedDevicesChanged();
                                }
                            }
                            catch (Exception parseEx)
                            {
                                Logger.Error("DeviceConnectionManager", $"解析客户端设备型号失败: {parseEx.Message}", parseEx);
                            }

                            // 回发服务端（PC）的设备型号：SWEC_PCDEVICE {"manufacturer":"<厂商>","model":"<型号>"}
                            var (serverManufacturer, serverModel) = GetServerDeviceModel();
                            var serverPayload = JsonConvert.SerializeObject(new
                            {
                                manufacturer = serverManufacturer,
                                model = serverModel
                            });
                            await SendLineAsync(stream, $"SWEC_PCDEVICE {serverPayload}");
                            Logger.Info("DeviceConnectionManager", $"发送服务端设备型号: SWEC_PCDEVICE {serverPayload}");
                        }
                        else
                        {
                            Logger.Info("DeviceConnectionManager", $"收到消息: {message}");
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                Logger.Error("DeviceConnectionManager", $"处理客户端连接失败: {ex.Message}", ex);
                ConnectionStatusChanged?.Invoke("连接断开，等待新设备连接...");
            }
            finally
            {
                // 仅在握手成功计入过计数时才减少
                if (handshakeSuccess)
                {
                    DecrementConnectedCount();
                    bool removed = false;
                    lock (_devicesLock)
                    {
                        removed = _connectedDevices.Remove(connectionId);
                    }
                    if (removed)
                    {
                        RaiseConnectedDevicesChanged();
                    }
                }
                client.Close();
            }
        }

        /// <summary>
        /// 获取服务端（本机 PC）的设备型号信息。
        /// 通过 WMI Win32_ComputerSystem 读取 Manufacturer 与 Model 字段。
        /// 读取失败时返回 "Unknown"。
        /// </summary>
        private (string Manufacturer, string Model) GetServerDeviceModel()
        {
            try
            {
                using var searcher = new ManagementObjectSearcher("SELECT Manufacturer, Model FROM Win32_ComputerSystem");
                foreach (var obj in searcher.Get())
                {
                    string manufacturer = obj["Manufacturer"]?.ToString()?.Trim() ?? "";
                    string model = obj["Model"]?.ToString()?.Trim() ?? "";
                    if (string.IsNullOrEmpty(manufacturer)) manufacturer = "Unknown";
                    if (string.IsNullOrEmpty(model)) model = "Unknown";
                    return (manufacturer, model);
                }
            }
            catch (Exception ex)
            {
                Logger.Error("DeviceConnectionManager", $"获取服务端设备型号失败: {ex.Message}", ex);
            }
            return ("Unknown", "Unknown");
        }

        /// <summary>
        /// 生成连接码
        /// </summary>
        public string GenerateConnectionCode()
        {
            _connectionCode = Guid.NewGuid().ToString("N").Substring(0, 8).ToUpper();
            Logger.Info("DeviceConnectionManager", $"生成连接码: {_connectionCode}");
            return _connectionCode;
        }

        /// <summary>
        /// 生成二维码
        /// </summary>
        public void GenerateQrCode()
        {
            try
            {
                var ipAddress = GetLocalIPAddress();
                // 局域网模式：JSON 格式，头部带 type 字段以标识当前模式
                var qrContent = $"{{\"type\":\"lan\",\"ip\":\"{ipAddress}\",\"port\":{_port}}}";
                
                var generator = new QRCodeGenerator();
                var data = generator.CreateQrCode(qrContent, QRCodeGenerator.ECCLevel.Q);
                var qrCode = new QRCode(data);

                // 从当前主题资源读取二维码颜色，这样无论启动主题还是运行中切换主题都能正确匹配
                Color moduleColor = Color.Black;
                Color backgroundColor = Color.White;
                try
                {
                    var mainWindow = System.Windows.Application.Current?.MainWindow;
                    if (mainWindow?.TryFindResource("QrModuleColor") is System.Windows.Media.Color m)
                    {
                        moduleColor = Color.FromArgb(m.A, m.R, m.G, m.B);
                    }
                    if (mainWindow?.TryFindResource("QrBackgroundColor") is System.Windows.Media.Color b)
                    {
                        backgroundColor = Color.FromArgb(b.A, b.R, b.G, b.B);
                    }
                }
                catch
                {
                    // 资源缺失时使用默认浅色主题配色
                }

                var barcodeBitmap = qrCode.GetGraphic(10, moduleColor, backgroundColor, false);

                _qrCodeImage = new BitmapImage();
                using (var stream = new System.IO.MemoryStream())
                {
                    barcodeBitmap.Save(stream, System.Drawing.Imaging.ImageFormat.Png);
                    stream.Position = 0;
                    _qrCodeImage.BeginInit();
                    _qrCodeImage.CacheOption = BitmapCacheOption.OnLoad;
                    _qrCodeImage.StreamSource = stream;
                    _qrCodeImage.EndInit();
                    _qrCodeImage.Freeze();
                }

                Logger.Info("DeviceConnectionManager", $"二维码生成成功: {qrContent}");
            }
            catch (Exception ex)
            {
                Logger.Error("DeviceConnectionManager", $"生成二维码失败: {ex.Message}", ex);
            }
        }

        /// <summary>
        /// 获取本机IP地址
        /// </summary>
        public string GetLocalIPAddress()
        {
            try
            {
                // 获取所有可用网络接口
                var interfaces = GetAvailableNetworkInterfaces();

                // 如果没有找到网络接口，返回默认值
                if (!interfaces.Any())
                {
                    return "127.0.0.1";
                }

                // 如果指定了网卡，优先使用该网卡。
                // 注意：查找时使用包含虚拟网卡的完整列表，因为用户可能已通过
                // "显示虚拟网卡"勾选了虚拟网卡，过滤列表会导致 Id 查不到而返回错误的兜底 IP。
                if (!string.IsNullOrEmpty(_selectedAdapterId))
                {
                    var allInterfaces = GetAvailableNetworkInterfaces(false);
                    var selected = allInterfaces.FirstOrDefault(ni => ni.Id == _selectedAdapterId);
                    if (selected != null)
                    {
                        var ip = GetIPv4FromInterface(selected);
                        if (ip != null) return ip;
                    }
                }

                // 优先获取有网关的网卡（当前正在使用的）
                foreach (var netInterface in interfaces)
                {
                    var ipProps = netInterface.GetIPProperties();
                    var gatewayAddresses = ipProps.GatewayAddresses;

                    if (gatewayAddresses != null && gatewayAddresses.Any())
                    {
                        foreach (var addrInfo in ipProps.UnicastAddresses)
                        {
                            if (addrInfo.Address.AddressFamily == AddressFamily.InterNetwork)
                            {
                                return addrInfo.Address.ToString();
                            }
                        }
                    }
                }

                // 如果没有找到有网关的网卡，返回第一个可用的IP
                foreach (var netInterface in interfaces)
                {
                    var ip = GetIPv4FromInterface(netInterface);
                    if (ip != null) return ip;
                }

                return "127.0.0.1";
            }
            catch (Exception ex)
            {
                Logger.Error("DeviceConnectionManager", $"获取IP地址失败: {ex.Message}", ex);
                return "127.0.0.1";
            }
        }

        /// <summary>
        /// 获取可用的网络接口列表。
        /// excludeVirtual=true 时过滤虚拟网卡（用于默认网卡选择等场景），false 时仅过滤环回。
        /// </summary>
        private List<System.Net.NetworkInformation.NetworkInterface> GetAvailableNetworkInterfaces(bool excludeVirtual = true)
        {
            return System.Net.NetworkInformation.NetworkInterface.GetAllNetworkInterfaces()
                .Where(ni => ni.OperationalStatus == System.Net.NetworkInformation.OperationalStatus.Up &&
                          ni.NetworkInterfaceType != System.Net.NetworkInformation.NetworkInterfaceType.Loopback &&
                          (!excludeVirtual ||
                           (!ni.Description.Contains("Virtual") &&
                            !ni.Description.Contains("Hyper-V") &&
                            !ni.Description.Contains("VMware") &&
                            !ni.Description.Contains("VirtualBox") &&
                            !ni.Description.Contains("Tunnel"))))
                .ToList();
        }

        /// <summary>
        /// 从指定网卡获取 IPv4 地址
        /// </summary>
        private string GetIPv4FromInterface(System.Net.NetworkInformation.NetworkInterface netInterface)
        {
            var ipProps = netInterface.GetIPProperties();
            foreach (var addrInfo in ipProps.UnicastAddresses)
            {
                if (addrInfo.Address.AddressFamily == AddressFamily.InterNetwork)
                {
                    return addrInfo.Address.ToString();
                }
            }
            return null;
        }

        /// <summary>
        /// 获取可用网卡列表，返回 (Id, 显示名称) 元组列表。
        /// excludeVirtual=true 过滤虚拟网卡，false 包含虚拟网卡。
        /// </summary>
        public List<(string Id, string DisplayName)> GetNetworkAdapters(bool excludeVirtual = true)
        {
            var result = new List<(string Id, string DisplayName)>();
            try
            {
                foreach (var ni in GetAvailableNetworkInterfaces(excludeVirtual))
                {
                    var ip = GetIPv4FromInterface(ni);
                    // 显示名称：网卡名 + IP 地址
                    string displayName = string.IsNullOrEmpty(ip)
                        ? ni.Name
                        : $"{ni.Name} ({ip})";
                    result.Add((ni.Id, displayName));
                }
            }
            catch (Exception ex)
            {
                Logger.Error("DeviceConnectionManager", $"获取网卡列表失败: {ex.Message}", ex);
            }
            return result;
        }
    }
